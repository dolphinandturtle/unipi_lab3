\noindent\colorbox{blu}{
  \begin{minipage}{1.0\linewidth}
    \paragraph{Risposta (a)}
    Qualitativamente il circuito si tratta di un amplificatore pilotato
    dall'ingresso non invertente che viene preceduto da un nodo che
    ridirige il flusso di corrente ad un blocco secondario da studiare
    qualitativamente. Nei limiti di frequenze il sotto-circuito agisce
    come se non esistesse. Il ramo in retroazione dell'ingresso invertente
    agisce come un \textit{dimezzatore di tensione}, dunque se l'impedenza
    effettiva del sotto-circuito equivale alla configurazione all'ingresso
    invertente il guadagno differenziale sar\'a nullo e il filtro agira
    come \textit{notch-filter}. Con uno studio quantitativo si potra calcolare
    quanti notch sono presenti e dove sono disposti in funzione della frequenza.
  \end{minipage}
} \smallskip \\

    \item[ii]

    \end{itemize}

\paragraph{Risposta (b.i)}
\begin{itemize}
\item
  U2 \'e un voltage follower e di conseguenza $V_{+U2} = V_{U2}$.
\item
  U2 non assorbe corrente per ipotesi di idealit\'a degli opamp, di conseguenza $\boldsymbol{I_1} = I_{R_4}$,
  da cui si ottiene $V_{+U2} - V_{R_4 \to R_5} = R_4\boldsymbol{I_1}$.
\end{itemize}
da cui si ottiene la prima relazione fondamentale che correla le tensioni $\boldsymbol{I_1}$ e $I_2$
\begin{equation*}
  I_2 = sC_2(V_{U2} - V_{R_4 \to R_5}) = sC_2(V_{+U2} - V_{R_4 \to R_5}) = sC_2R_4\boldsymbol{I_1} \implies \boldsymbol{I_2}(\boldsymbol{I_1})
\end{equation*}
da cui si ottiene direttamente il valore della tensione al nodo fra le resistenze $R_4$ e $R_5$
per causa diretta della legge di Kirchoff sui nodi:
\begin{equation*}
  V_{R_4 \to R_5} = R_5I_{R_5} = R_5(\boldsymbol{I_1} + \boldsymbol{I_2}) = R_5\boldsymbol{I_1} + sR_4R_5C_2\boldsymbol{I_1} \implies \boldsymbol{V_{R_4 \to R_5}}(\boldsymbol{I_1})
\end{equation*}
l'analisi rimanente da fare
\begin{equation*}
  \begin{gathered}
    \boldsymbol{I_1} = \frac{(V_{+U2} - \boldsymbol{V_{R_4\to R_5}})}{R_4} \implies V_{+U2} = (R_4 + R_5 + sR_4R_5C_2)\boldsymbol{I_1} \implies \boldsymbol{V_{+U2}}(\boldsymbol{I_1}) \\
    \boldsymbol{I_1} = \frac{(V_{+U1} - \boldsymbol{V_{+U2}})}{\frac{1}{sC_1}} \implies V_{+U1} = \frac{\boldsymbol{I_1}}{sC_1} + (R_4 + R_5 + sR_4R_5C_2)\boldsymbol{I_1} \implies \boldsymbol{V_{+U1}}(\boldsymbol{I_1}) \\
    \colorbox{blu}{\boxed{Z_{in} \equiv \frac{V_{+U1}(I_1)}{I_1} = \frac{1}{sC_1} + (R_4 + R_5 + sR_4R_5C_2)}}
  \end{gathered}
\end{equation*}
\paragraph{Risposta (b.ii)}

\begin{equation*}
  H(s) = \frac{Z_{in} - R}{Z_{in} + R}
\end{equation*}
\begin{equation}
  \label{eq:notch2-tfunc}
  \colorbox{blu}{\boxed{H(s) = \frac{1 + 25\xi^2}{1 + 2\xi + 25\xi^2} \qquad \xi \equiv sRC}}
\end{equation}
