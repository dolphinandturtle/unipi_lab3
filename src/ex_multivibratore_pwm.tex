\subsection*{Risposta (a)}
\begin{equation*}
  \frac{V_{OUT} - V_-}{Z_{R_3}} = \frac{V_- - V_M}{Z_{R_4}} + \frac{V_-}{Z_{C_1}}
\end{equation*}
sostituendo con i valori del circuito
\begin{equation*}
  V_- = \frac{V_{OUT} + V_M}{2 + sRC}
\end{equation*}
nell'ipotesi in cui l'op-amp sia ideale, l'ingresso \underline{non}-invertente non assorbe corrente
dunque la tensione \'e quella del partitore di tensione $Z_{R_1}-Z_{R_2}$
\begin{equation*}
  V_+ = \frac{1}{101}V_{OUT}
\end{equation*}
Si studia l'instabilit\'a del circuito mediante il fattore di retro-azione derivabile dalla tensione
d'ingresso (differenziale)
\begin{equation*}
  V_d = V_+ - V_-
\end{equation*}
\begin{equation*}
  V_{d} = \left(\frac{sRC - 99}{101(2 + sRC)}\right)V_{OUT} - \frac{V_M}{2 + sRC}
\end{equation*}
\begin{equation*}
  \beta(s) \equiv \frac{sRC - 99}{101(2 + sRC)}
\end{equation*}
\begin{itemize}
\item
  Quando il condensatore \'e completamente scarico il suo ramo ``corto-circuita'' la terra
  \begin{equation*}
    V_d = \frac{1}{101}V_{OUT}
  \end{equation*}
  dunque il fattore di retro-azione \'e positivo e di conseguenza il circuito non \'e lineare
\item
  Quando il condensatore \'e pienamente carico il suo ramo si ``apre''
  \begin{equation*}
    V_d = -\frac{99}{202}V_{OUT} - \frac12 V_M
  \end{equation*}
  dunque il fattore di retro-azione \'e negativo e di conseguenza il circuito pu\'o essere lineare.
\end{itemize}
Durante il regime di linearit\'a
\begin{equation*}
  V_{OUT} = -\frac{101}{99 - sRC}V_M
\end{equation*}

\begin{equation*}
  \begin{gathered}
    \frac{V_{OUT} - V_-}{100R} = \frac{V_- - V_M}{100R} + 100C\dot{V}_- \\
    V_{OUT} - V_- = V_- - V_M + 10^4RC\dot{V}_-
  \end{gathered}
\end{equation*}

\paragraph{Regime non-lineare}
\begin{equation*}
  V_d > 0 \implies \frac{1}{101}V_{OH} > 0
\end{equation*}
